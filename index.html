🚀 Benvenuto in Consumi Manager Cloud!

Questa applicazione salva i tuoi dati su Supabase per avere accesso ovunque e sincronizzazione in tempo reale.

1 Crea progetto Supabase

Vai su supabase.com, registrati e crea un nuovo progetto.

2 Esegui migrazione SQL

Nel dashboard Supabase, vai su SQL Editor e crea una nuova query. Copia e incolla questo script:

-- Consumi Manager - Database Migration
CREATE TABLE IF NOT EXISTS forniture (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  codice TEXT UNIQUE NOT NULL,
  nome TEXT NOT NULL,
  attiva BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS contatori (
  id SERIAL PRIMARY KEY,
  nome TEXT NOT NULL,
  codice TEXT UNIQUE NOT NULL,
  fornitura TEXT NOT NULL,
  ubicazione TEXT,
  stato TEXT DEFAULT 'attivo' CHECK (stato IN ('attivo', 'inattivo')),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS consumi (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  tipo TEXT NOT NULL CHECK (tipo IN ('energia', 'gas', 'acqua')),
  contatore_id INTEGER REFERENCES contatori(id) ON DELETE CASCADE,
  mese TEXT NOT NULL,
  consumo NUMERIC(10,2) NOT NULL,
  consumo_originale NUMERIC(10,2),
  tipo_origine TEXT,
  importo NUMERIC(10,2) DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS allegati (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  consumo_id UUID REFERENCES consumi(id) ON DELETE CASCADE,
  nome TEXT NOT NULL,
  tipo_file TEXT,
  data TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS configurazioni (
  chiave TEXT PRIMARY KEY,
  valore TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_consumi_tipo ON consumi(tipo);
CREATE INDEX IF NOT EXISTS idx_consumi_contatore ON consumi(contatore_id);
CREATE INDEX IF NOT EXISTS idx_consumi_mese ON consumi(mese);
CREATE INDEX IF NOT EXISTS idx_allegati_consumo ON allegati(consumo_id);
CREATE INDEX IF NOT EXISTS idx_contatori_fornitura ON contatori(fornitura);

ALTER TABLE forniture ENABLE ROW LEVEL SECURITY;
ALTER TABLE contatori ENABLE ROW LEVEL SECURITY;
ALTER TABLE consumi ENABLE ROW LEVEL SECURITY;
ALTER TABLE allegati ENABLE ROW LEVEL SECURITY;
ALTER TABLE configurazioni ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Enable all for authenticated users" ON forniture FOR ALL USING (true);
CREATE POLICY "Enable all for authenticated users" ON contatori FOR ALL USING (true);
CREATE POLICY "Enable all for authenticated users" ON consumi FOR ALL USING (true);
CREATE POLICY "Enable all for authenticated users" ON allegati FOR ALL USING (true);
CREATE POLICY "Enable all for authenticated users" ON configurazioni FOR ALL USING (true);

INSERT INTO forniture (codice, nome, attiva) VALUES
  ('energia', 'Energia Elettrica', true),
  ('gas', 'Gas', true),
  ('acqua', 'Acqua', true)
ON CONFLICT (codice) DO NOTHING;

INSERT INTO contatori (nome, codice, fornitura, ubicazione, stato) VALUES
  ('Casa principale', 'EN001', 'energia', 'Abitazione', 'attivo'),
  ('Riscaldamento', 'GS001', 'gas', 'Caldaia', 'attivo'),
  ('Contatore acqua', 'AQ001', 'acqua', 'Generale', 'attivo')
ON CONFLICT (codice) DO NOTHING;

Dopo aver incollato, clicca Run per eseguire lo script.
